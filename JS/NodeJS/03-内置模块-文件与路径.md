# 内置模块：文件系统与路径

## fs (File System)

### 读取文件

```javascript
const fs = require('fs')

// 同步读取
const data = fs.readFileSync('./file.txt', 'utf-8')
console.log(data)

// 异步回调
fs.readFile('./file.txt', 'utf-8', (err, data) => {
  if (err) throw err
  console.log(data)
})

// Promise 形式 (fs.promises)
async function read() {
  const data = await fs.promises.readFile('./file.txt', 'utf-8')
  console.log(data)
}
```

### 写入文件

```javascript
// 覆盖写入
fs.writeFileSync('./file.txt', 'Hello World', 'utf-8')

// 追加写入
fs.appendFileSync('./file.txt', '\nNew line', 'utf-8')

// Promise 形式
await fs.promises.writeFile('./file.txt', 'content')
```

### 文件信息

```javascript
const stats = fs.statSync('./file.txt')
stats.isFile()       // true
stats.isDirectory()  // false
stats.size           // 字节数
stats.birthtime      // 创建时间
stats.mtime          // 修改时间
```

### 目录操作

```javascript
// 创建目录
fs.mkdirSync('./new-dir', { recursive: true })

// 读取目录
const files = fs.readdirSync('./dir')
// ['file1.txt', 'file2.js']

// 删除目录（需为空）
fs.rmdirSync('./empty-dir')

// 递归删除
fs.rmSync('./dir', { recursive: true, force: true })

// 检查路径是否存在
fs.existsSync('./file.txt')
```

### 文件流 (大文件读写)

```javascript
const readStream = fs.createReadStream('./large.txt', 'utf-8')
const writeStream = fs.createWriteStream('./copy.txt')

readStream.on('data', chunk => {
  console.log('Received chunk:', chunk.length)
})

readStream.pipe(writeStream) // 管道传输
```

### 监听文件变化

```javascript
fs.watch('./file.txt', (eventType, filename) => {
  console.log(`文件 ${filename}: ${eventType}`)
})
```

## path (路径处理)

### 常用方法

```javascript
const path = require('path')

// 路径拼接（自动处理分隔符）
path.join('/base', 'dir', 'file.txt')
// /base/dir/file.txt

// 解析为绝对路径
path.resolve('dir', 'file.txt')
// /current/working/dir/file.txt

// 获取目录名
path.dirname('/a/b/c.txt')
// /a/b

// 获取文件名（含扩展名）
path.basename('/a/b/c.txt')
// c.txt

// 获取扩展名
path.extname('/a/b/c.txt')
// .txt

// 不含扩展名的文件名
path.basename('/a/b/c.txt', '.txt')
// c

// 解析路径各组成部分
path.parse('/a/b/c.txt')
// { root: '/', dir: '/a/b', base: 'c.txt', ext: '.txt', name: 'c' }

// 格式化路径对象为字符串
path.format({ dir: '/a/b', base: 'c.txt' })
// /a/b/c.txt

// 判断是否为绝对路径
path.isAbsolute('/a/b')  // true
path.isAbsolute('a/b')   // false

// 获取相对路径
path.relative('/data/a', '/data/b/c.txt')
// ../b/c.txt
```

### 路径分隔符

```javascript
path.sep       // Linux: '/'   Windows: '\\'
path.delimiter // Linux: ':'   Windows: ';'

// POSIX 与 Win32 独立版本
path.posix.join(...)
path.win32.join(...)
```

## 实用组合示例

```javascript
const fs = require('fs')
const path = require('path')

// 遍历目录
function walkDir(dir) {
  const entries = fs.readdirSync(dir, { withFileTypes: true })
  for (const entry of entries) {
    const fullPath = path.join(dir, entry.name)
    if (entry.isDirectory()) {
      walkDir(fullPath)
    } else {
      console.log(fullPath)
    }
  }
}

// 确保目录存在
function ensureDir(dirPath) {
  if (!fs.existsSync(dirPath)) {
    fs.mkdirSync(dirPath, { recursive: true })
  }
}
```
