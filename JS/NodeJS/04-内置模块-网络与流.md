# 内置模块：网络与流

## http / https

### 创建 HTTP 服务器

```javascript
const http = require('http')

const server = http.createServer((req, res) => {
  // req: IncomingMessage  (可读流)
  // res: ServerResponse   (可写流)

  res.statusCode = 200
  res.setHeader('Content-Type', 'text/plain; charset=utf-8')
  res.end('Hello World\n')
})

server.listen(3000, () => {
  console.log('Server running at http://localhost:3000/')
})
```

### 请求对象 (req)

```javascript
const server = http.createServer((req, res) => {
  console.log(req.method)      // GET, POST, ...
  console.log(req.url)         // /path?query=string
  console.log(req.headers)     // 请求头对象
  console.log(req.httpVersion) // 1.1

  // 读取请求体
  let body = ''
  req.on('data', chunk => body += chunk)
  req.on('end', () => console.log('Body:', body))
})
```

### 响应对象 (res)

```javascript
res.writeHead(200, {
  'Content-Type': 'application/json',
  'X-Custom-Header': 'value'
})
res.end(JSON.stringify({ message: 'ok' }))

// 分块发送
res.write('第一块')
res.write('第二块')
res.end()
```

### 发送 HTTP 请求

```javascript
// GET 请求
http.get('http://api.example.com/data', res => {
  let data = ''
  res.on('data', chunk => data += chunk)
  res.on('end', () => console.log(JSON.parse(data)))
})

// POST 请求
const data = JSON.stringify({ name: 'Alice' })
const options = {
  hostname: 'api.example.com',
  port: 80,
  path: '/users',
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Content-Length': Buffer.byteLength(data)
  }
}
const req = http.request(options, res => {
  // 处理响应
})
req.write(data)
req.end()
```

### URL 解析

```javascript
const url = require('url')
const parsed = new URL('http://user:pass@site.com:8080/path?q=1#hash')
// { protocol: 'http:', hostname: 'site.com', port: '8080', ... }
```

## Buffer (二进制数据)

```javascript
// 创建 Buffer
Buffer.from('hello')                    // UTF-8 编码
Buffer.from('hello', 'base64')          // Base64 解码
Buffer.from([0x68, 0x65, 0x6c, 0x6c, 0x6f])
Buffer.alloc(10)                        // 分配 10 字节（填0）
Buffer.allocUnsafe(10)                  // 分配 10 字节（不初始化）

// 读写
const buf = Buffer.from('Hello')
buf[0]                  // 72 (0x48)
buf.length              // 5
buf.toString()          // 'Hello'
buf.toString('hex')     // '48656c6c6f'
buf.toString('base64')  // 'SGVsbG8='

buf.write('Hi')
buf.slice(0, 2)

// 合并
Buffer.concat([buf1, buf2])
```

## Stream (流)

四种流类型：

| 类型 | 说明 | 示例 |
|------|------|------|
| Readable | 可读流 | `fs.createReadStream()` |
| Writable | 可写流 | `fs.createWriteStream()` |
| Duplex | 双工流 | `net.Socket` |
| Transform | 转换流 | `zlib.createGzip()` |

### 可读流

```javascript
const stream = fs.createReadStream('./file.txt', {
  highWaterMark: 64 * 1024, // 每次读取 64KB
  encoding: 'utf-8'
})

stream.on('data', chunk => console.log(chunk))
stream.on('end', () => console.log('读取完成'))
stream.on('error', err => console.error(err))

// 暂停/恢复
stream.pause()
stream.resume()
```

### 可写流

```javascript
const stream = fs.createWriteStream('./output.txt')

stream.write('第一行\n')
stream.write('第二行\n')
stream.end('最后一行\n')

stream.on('finish', () => console.log('写入完成'))
stream.on('error', err => console.error(err))
```

### 管道 (Pipe)

```javascript
// 文件复制
readStream.pipe(writeStream)

// 压缩
const zlib = require('zlib')
readStream.pipe(zlib.createGzip()).pipe(writeStream)

// HTTP 响应流式传输
http.createServer((req, res) => {
  const stream = fs.createReadStream('./large-file.mp4')
  stream.pipe(res)
})
```

### 自定义流 (Transform)

```javascript
const { Transform } = require('stream')

const upperCase = new Transform({
  transform(chunk, encoding, callback) {
    this.push(chunk.toString().toUpperCase())
    callback()
  }
})

process.stdin.pipe(upperCase).pipe(process.stdout)
```

## zlib (压缩/解压)

```javascript
const zlib = require('zlib')
const fs = require('fs')

// gzip 压缩
fs.createReadStream('input.txt')
  .pipe(zlib.createGzip())
  .pipe(fs.createWriteStream('input.txt.gz'))

// gzip 解压
fs.createReadStream('input.txt.gz')
  .pipe(zlib.createGunzip())
  .pipe(fs.createWriteStream('output.txt'))
```
