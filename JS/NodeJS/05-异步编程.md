# 异步编程

Node.js 异步编程的四种主要模式：**Callback → Promise → async/await → EventEmitter**。

---

## 回调 (Callback)

最早期的异步模式。**回调地狱**是其主要问题。

```javascript
const fs = require('fs')

// Node.js 回调风格：error-first
fs.readFile('./a.txt', 'utf-8', (err, data) => {
  if (err) {
    console.error('读取失败:', err)
    return
  }
  console.log(data)
})

// 回调地狱：嵌套过深
fs.readFile('a.txt', (err, dataA) => {
  fs.readFile('b.txt', (err, dataB) => {
    fs.writeFile('c.txt', dataA + dataB, (err) => {
      console.log('完成')
    })
  })
})
```

### 回调约定 (Error-First Callback)

```javascript
/**
 * @param {Error|null} err  - 错误对象，成功时为 null
 * @param {*}          data - 成功时的数据
 */
function myAsyncFn(callback) {
  setTimeout(() => {
    try {
      const result = doSomething()
      callback(null, result)
    } catch (err) {
      callback(err)
    }
  }, 100)
}
```

## Promise

ES6 引入，解决回调地狱问题。

### 基本用法

```javascript
const promise = new Promise((resolve, reject) => {
  // 异步操作成功
  resolve(value)
  // 异步操作失败
  reject(new Error('fail'))
})

promise
  .then(value => console.log(value))
  .catch(err => console.error(err))
  .finally(() => console.log('done'))
```

### 链式调用

```javascript
readFile('a.txt')
  .then(data => readFile('b.txt'))
  .then(data => writeFile('c.txt', data))
  .then(() => console.log('完成'))
  .catch(err => console.error(err))
```

### 静态方法

```javascript
// 全部完成
Promise.all([p1, p2, p3])
Promise.allSettled([p1, p2, p3]) // 不短路，返回所有结果

// 竞速
Promise.race([p1, p2])       // 第一个完成
Promise.any([p1, p2])        // 第一个兑现（忽略拒绝）

// 快速创建
Promise.resolve('value')
Promise.reject(new Error('err'))

// 顺序执行（手写）
async function sequential(tasks) {
  const results = []
  for (const task of tasks) {
    results.push(await task())
  }
  return results
}
```

### 回调转 Promise

```javascript
const fs = require('fs')
const { promisify } = require('util')

const readFile = promisify(fs.readFile)
const data = await readFile('file.txt', 'utf-8')

// Node.js 10+ 内置 promisify
// fs.promises.*
```

### 常见错误

```javascript
// ❌ 忘记 return
new Promise(resolve => {
  resolve(1)
})
.then(result => {
  new Promise(resolve => resolve(result + 1)) // 没返回
})
.then(result => console.log(result)) // undefined

// ✅ 正确
.then(result => {
  return new Promise(resolve => resolve(result + 1))
})
```

## async/await

ES2017 引入，Promise 的语法糖。

```javascript
async function main() {
  try {
    const dataA = await readFile('a.txt')
    const dataB = await readFile('b.txt')
    await writeFile('c.txt', dataA + dataB)
    console.log('完成')
  } catch (err) {
    console.error('失败:', err)
  }
}
```

### 并行执行

```javascript
// 顺序（慢）
const a = await readFile('a.txt')
const b = await readFile('b.txt')

// 并行（快）
const [a, b] = await Promise.all([
  readFile('a.txt'),
  readFile('b.txt')
])
```

### 顶层 await (ESM)

```javascript
// 仅 ES Modules (.mjs / "type": "module")
const data = await fetch('https://api.example.com')
console.log(data)
```

### 遍历中的异步

```javascript
// for...of 顺序执行
for (const url of urls) {
  const data = await fetch(url)
}

// Promise.all 并行
const results = await Promise.all(urls.map(fetch))

// 限流并发
async function mapLimit(arr, limit, fn) {
  const results = []
  const queue = []
  for (const item of arr) {
    const p = fn(item).then(r => results.push(r))
    queue.push(p)
    if (queue.length >= limit) {
      await Promise.race(queue)
      queue.splice(queue.findIndex(p => p._settled))
    }
  }
  await Promise.all(queue)
  return results
}
```

## EventEmitter (事件触发器)

Node.js 的核心模式，许多内置模块继承自 EventEmitter。

### 基本用法

```javascript
const EventEmitter = require('events')

// 创建
const emitter = new EventEmitter()

// 监听
emitter.on('event', (arg1, arg2) => {
  console.log('事件触发:', arg1, arg2)
})

// 单次监听
emitter.once('event', () => {
  console.log('只触发一次')
})

// 触发
emitter.emit('event', 'a', 'b')
```

### 自定义 EventEmitter 类

```javascript
const { EventEmitter } = require('events')

class MyClass extends EventEmitter {
  start() {
    this.emit('start', Date.now())
    setInterval(() => {
      this.emit('data', { timestamp: Date.now() })
    }, 1000)
  }

  stop() {
    this.emit('end')
  }
}

const mc = new MyClass()
mc.on('data', data => console.log(data))
mc.on('end', () => console.log('stopped'))
mc.start()
```

### 常用方法

```javascript
emitter.on(event, listener)       // 添加监听器
emitter.once(event, listener)     // 添加单次监听器
emitter.emit(event, ...args)      // 触发事件
emitter.removeListener(event, fn) // 移除指定监听器
emitter.removeAllListeners(event) // 移除所有监听器
emitter.listeners(event)          // 获取监听器数组
emitter.listenerCount(event)      // 监听器数量
emitter.eventNames()              // 所有已注册事件名
emitter.setMaxListeners(n)        // 设置最大监听器数（避免内存泄漏警告）
```

### 错误事件

```javascript
const emitter = new EventEmitter()

// 未监听 'error' 时触发会导致进程崩溃
emitter.on('error', err => {
  console.error('捕获错误:', err.message)
})

emitter.emit('error', new Error('something broke'))
```

### 事件流 (Stream + EventEmitter 结合)

```javascript
const stream = fs.createReadStream('file.txt')
stream.on('data', chunk => {})  // 继承自 EventEmitter
stream.on('end', () => {})
stream.on('error', err => {})
```

## 事件循环 (Event Loop)

Node.js 事件循环阶段：

```
   ┌───────────────────────────┐
┌─>│           timers          │ ← setTimeout/setInterval
│  └─────────────┬─────────────┘
│  ┌─────────────┴─────────────┐
│  │     pending callbacks     │ ← 延迟回调
│  └─────────────┬─────────────┘
│  ┌─────────────┴─────────────┐
│  │       idle, prepare       │ ← 内部
│  └─────────────┬─────────────┘
│  ┌─────────────┴─────────────┐
│  │           poll            │ ← I/O 事件
│  └─────────────┬─────────────┘
│  ┌─────────────┴─────────────┐
│  │           check           │ ← setImmediate
│  └─────────────┬─────────────┘
│  ┌─────────────┴─────────────┐
└──┤      close callbacks      │ ← close 事件
   └───────────────────────────┘
```

### process.nextTick vs setImmediate

```javascript
// nextTick 在当前阶段结束后立即执行（优先级最高）
process.nextTick(() => console.log('nextTick'))

// setImmediate 在 check 阶段执行
setImmediate(() => console.log('setImmediate'))

// setTimeout(fn, 0) ≈ setImmediate（但顺序不确定）
setTimeout(() => console.log('timeout'), 0)
```
