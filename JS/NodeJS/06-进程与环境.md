# 进程与环境

## process (进程对象)

无需引入，全局可用。

### 进程信息

```javascript
// 命令行参数
process.argv
// ['node', '/path/app.js', '--port', '3000']

// 工作目录
process.cwd()        // 当前工作目录
process.chdir('/tmp') // 切换工作目录

// 内存使用
process.memoryUsage()
// { rss, heapTotal, heapUsed, external, arrayBuffers }

// CPU 使用
process.cpuUsage()
// { user, system }

// 平台信息
process.platform   // 'linux' | 'darwin' | 'win32'
process.arch       // 'x64' | 'arm64'
process.version    // Node.js 版本
process.versions   // 各依赖版本
process.release    // 发布信息
process.features   // 编译特性

// 进程 ID
process.pid

// 运行时间
process.uptime()   // 秒
```

### 环境变量

```javascript
// 读取
process.env.NODE_ENV
process.env.PATH

// 设置（仅当前进程生效）
process.env.MY_VAR = 'value'

// 常见环境变量
// NODE_ENV: 'development' | 'production'
// PORT: 端口号

// .env 文件（配合 dotenv 包）
// require('dotenv').config()
```

### 进程控制

```javascript
// 退出
process.exit(0)      // 成功
process.exit(1)      // 失败
process.exitCode = 1 // 设置退出码但不立即退出

// 信号
process.on('SIGINT', () => {
  console.log('收到 SIGINT')
  process.exit(0)
})
process.on('SIGTERM', () => {
  console.log('收到 SIGTERM')
  process.exit(0)
})

// 当前进程不可杀死
// process.on('SIGKILL', ...) // 不能监听
```

### 标准流

```javascript
process.stdin   // 可读流
process.stdout  // 可写流
process.stderr  // 可写流

// 读取输入
process.stdin.setEncoding('utf-8')
process.stdin.on('data', input => {
  console.log('输入:', input.trim())
})

// 输出
process.stdout.write('Hello\n')
process.stderr.write('Error\n')

// 与 console 等价
// console.log = process.stdout.write + '\n'
// console.error = process.stderr.write + '\n'
```

### 进程事件

```javascript
process.on('exit', code => {
  // 只能执行同步操作
  console.log('进程退出码:', code)
})

process.on('beforeExit', code => {
  // 事件循环清空后触发
})

process.on('uncaughtException', err => {
  // 捕获未处理的异常（最后防线）
  console.error('未捕获异常:', err)
  process.exit(1)
})

process.on('unhandledRejection', (reason, promise) => {
  // 未处理的 Promise 拒绝
  console.error('未处理的 Promise 拒绝:', reason)
})

process.on('warning', warning => {
  // 进程警告
  console.warn(warning.name, warning.message)
})
```

### 高级操作

```javascript
// 设置进程标题（ps 可见）
process.title = 'my-app'

// 高精度时间
const start = process.hrtime.bigint()
// ... 操作 ...
const end = process.hrtime.bigint()
console.log(`耗时: ${(end - start) / 1_000_000n}ms`)

// 资源使用（类似 ulimit）
// process.resourceUsage()

// 获取进程组 ID / 会话 ID
process.getgid()
process.getuid()
process.getgroups()
```

## os (操作系统)

```javascript
const os = require('os')

// 系统信息
os.hostname()        // 主机名
os.type()            // 'Linux' | 'Darwin' | 'Windows_NT'
os.platform()        // 'linux' | 'darwin' | 'win32'
os.release()         // 内核版本
os.arch()            // 'x64' | 'arm64'

// CPU 信息
os.cpus()            // 每个 CPU 核心信息（型号、速度）
os.cpus().length     // 逻辑 CPU 核心数
os.freemem()         // 空闲内存（字节）
os.totalmem()        // 总内存（字节）
os.loadavg()         // 1/5/15 分钟平均负载（仅 Unix）

// 网络
os.networkInterfaces() // 网络接口信息
os.homedir()         // 用户主目录
os.tmpdir()          // 临时目录
os.EOL               // 换行符（\n 或 \r\n）
os.uptime()          // 系统运行时间（秒）
os.userInfo()        // 当前用户信息
```

### systemd 或 docker 环境检测

```javascript
const isDocker = os.hostname() !== require('os').hostname()
// 或检查 /proc/1/cgroup
```

## child_process (子进程)

### spawn - 流式执行

```javascript
const { spawn } = require('child_process')

// 执行命令
const child = spawn('ls', ['-la', '/tmp'], {
  cwd: '/tmp',
  env: { ...process.env, MY_VAR: 'val' },
  stdio: 'pipe',     // 'inherit' | 'pipe' | 'ignore'
  shell: false       // true 则通过 shell 执行
})

child.stdout.on('data', data => console.log(data.toString()))
child.stderr.on('data', data => console.error(data.toString()))
child.on('close', code => console.log('exit code:', code))
child.on('error', err => console.error('spawn error:', err))
```

### exec - 缓冲执行

```javascript
const { exec } = require('child_process')

exec('ls -la', (err, stdout, stderr) => {
  if (err) {
    console.error(`执行错误: ${err}`)
    return
  }
  console.log(`stdout: ${stdout}`)
  console.error(`stderr: ${stderr}`)
})

// 超时控制
exec('sleep 10', { timeout: 5000 }, (err) => {
  if (err.killed) console.log('超时被终止')
})
```

### execFile - 直接执行文件（不经过 shell）

```javascript
const { execFile } = require('child_process')
execFile('node', ['--version'], (err, stdout) => {
  console.log(stdout)
})
```

### fork - 创建 Node.js 子进程

```javascript
// master.js
const { fork } = require('child_process')
const child = fork('./worker.js')

child.send({ hello: 'world' })  // 发送 IPC 消息
child.on('message', msg => console.log('来自子进程:', msg))
child.on('exit', code => console.log('子进程退出:', code))

// worker.js
process.on('message', msg => {
  console.log('来自父进程:', msg)
  process.send({ received: true })
})
```

### 同步版本

```javascript
const { execSync, spawnSync } = require('child_process')

const stdout = execSync('ls -la', { encoding: 'utf-8' })
console.log(stdout)

// pipe 模式下返回子进程输出
const result = spawnSync('ls', ['-la'])
console.log(result.stdout.toString())
```

## cluster (集群)

利用多核 CPU 提升吞吐量。

```javascript
const cluster = require('cluster')
const http = require('http')
const os = require('os')

if (cluster.isMaster) {
  const cpuCount = os.cpus().length
  console.log(`Master ${process.pid} 启动`)

  // 为每个 CPU 创建 worker
  for (let i = 0; i < cpuCount; i++) {
    cluster.fork()
  }

  cluster.on('exit', (worker, code, signal) => {
    console.log(`Worker ${worker.process.pid} 退出`)
    // 自动重启
    cluster.fork()
  })

} else {
  // Worker 进程
  http.createServer((req, res) => {
    res.end(`Worker ${process.pid} 处理请求`)
  }).listen(8000)

  console.log(`Worker ${process.pid} 启动`)
}
```

### cluster 高级

```javascript
// 优雅关闭
cluster.on('exit', (worker) => {
  if (!worker.exitedAfterDisconnect) {
    cluster.fork() // 非预期退出才重启
  }
})

// 滚动重启
Object.values(cluster.workers).forEach(worker => {
  worker.disconnect()
  worker.on('disconnect', () => cluster.fork())
})
```
